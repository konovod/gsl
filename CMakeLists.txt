cmake_minimum_required(VERSION 3.22)

# ---------------------------------------------------------------------------
# Version information is read from configure.ac so the CMake build cannot
# drift from the autotools build.
# ---------------------------------------------------------------------------
file(READ "${CMAKE_CURRENT_SOURCE_DIR}/configure.ac" GSL_CONFIGURE_AC)

string(REGEX MATCH "AC_INIT\\(\\[gsl\\], *\\[([0-9.]+)\\]\\)" _gsl_ac_init "${GSL_CONFIGURE_AC}")
if(NOT CMAKE_MATCH_1)
  message(FATAL_ERROR "could not determine the GSL version from configure.ac")
endif()
set(GSL_VERSION "${CMAKE_MATCH_1}")

foreach(_var GSL_CURRENT GSL_REVISION GSL_AGE CBLAS_CURRENT CBLAS_REVISION CBLAS_AGE)
  string(REGEX MATCH "${_var}=([0-9]+)" _gsl_match "${GSL_CONFIGURE_AC}")
  if("${_gsl_match}" STREQUAL "")
    message(FATAL_ERROR "could not find ${_var} in configure.ac")
  endif()
  set(${_var} "${CMAKE_MATCH_1}")
endforeach()

project(GSL VERSION ${GSL_VERSION} LANGUAGES C)

# Libtool version-info C:R:A maps to a shared object version (C-A).A.R
math(EXPR GSL_SOVERSION "${GSL_CURRENT} - ${GSL_AGE}")
set(GSL_LIB_VERSION "${GSL_SOVERSION}.${GSL_AGE}.${GSL_REVISION}")
math(EXPR GSL_CBLAS_SOVERSION "${CBLAS_CURRENT} - ${CBLAS_AGE}")
set(GSL_CBLAS_LIB_VERSION "${GSL_CBLAS_SOVERSION}.${CBLAS_AGE}.${CBLAS_REVISION}")

message(STATUS "GSL version ${GSL_VERSION}")
message(STATUS "  libgsl soversion      : ${GSL_SOVERSION} (${GSL_LIB_VERSION})")
message(STATUS "  libgslcblas soversion : ${GSL_CBLAS_SOVERSION} (${GSL_CBLAS_LIB_VERSION})")

# Stage all public headers into <build>/include/gsl so that the sources'
# <gsl/...> includes resolve (the autotools build does this with symlinks).

set(GSL_HEADER_FILES "")
file(GLOB _gsl_top_headers "${CMAKE_CURRENT_SOURCE_DIR}/gsl*.h")
list(APPEND GSL_HEADER_FILES ${_gsl_top_headers})

file(GLOB _gsl_dirs LIST_DIRECTORIES true "${CMAKE_CURRENT_SOURCE_DIR}/*")
foreach(_dir IN LISTS _gsl_dirs)
  if(NOT IS_DIRECTORY "${_dir}")
    continue()
  endif()
  get_filename_component(_name "${_dir}" NAME)
  if(_name STREQUAL "gsl" OR _name STREQUAL "doc" OR _name STREQUAL "doc_texinfo"
     OR _name STREQUAL "build.vc" OR _name MATCHES "^\\.git")
    continue()
  endif()
  if(EXISTS "${_dir}/Makefile.am")
    file(GLOB _module_headers "${_dir}/gsl*.h")
    list(APPEND GSL_HEADER_FILES ${_module_headers})
  endif()
endforeach()

list(REMOVE_DUPLICATES GSL_HEADER_FILES)
list(FILTER GSL_HEADER_FILES EXCLUDE REGEX "gsl_version\\.h$")

foreach(_header IN LISTS GSL_HEADER_FILES)
  get_filename_component(_header_name "${_header}" NAME)
  configure_file("${_header}" "${GSL_STAGED_INCLUDE}/gsl/${_header_name}" COPYONLY)
endforeach()
list(LENGTH GSL_HEADER_FILES _gsl_num_headers)

function(gsl_configure_target target)
  set_target_properties(${target} PROPERTIES
    C_STANDARD 99
    C_STANDARD_REQUIRED ON)
  target_include_directories(${target} PRIVATE ${GSL_BUILD_INCLUDE_DIRS})
  target_compile_definitions(${target} PRIVATE HAVE_CONFIG_H)
endfunction()

gsl_configure_target(gslcblas)
gsl_configure_target(gsl)

# Consumers see the installed headers.

  "$<BUILD_INTERFACE:${GSL_STAGED_INCLUDE}>"
  "$<INSTALL_INTERFACE:${CMAKE_INSTALL_INCLUDEDIR}>")

  "$<BUILD_INTERFACE:${GSL_STAGED_INCLUDE}>"

    target_compile_definitions(gsl INTERFACE WIN32 GSL_DLL)
  else()

    target_compile_definitions(gsl INTERFACE WIN32)
  endif()

# ---------------------------------------------------------------------------
# Test programs (optional)
# ---------------------------------------------------------------------------
function(gsl_add_test module name)
  set(_sources ${ARGN})
  set(_target "${module}_${name}")
  add_executable(${_target} ${_sources})
  gsl_configure_target(${_target})
  target_link_libraries(${_target} PRIVATE gsl gslcblas)
  add_test(NAME "${module}.${name}" COMMAND ${_target}
           WORKING_DIRECTORY "${CMAKE_CURRENT_SOURCE_DIR}/${module}")

  # rng.test hardcodes expected values for the random32/random64 BSD and libc5
  # generators, which depend on the width of 'long'.  These expectations only
  # hold on some ABIs, so allow the test to be skipped where they do not.
  if(module STREQUAL "rng" AND name STREQUAL "test")
    set_tests_properties("${module}.${name}" PROPERTIES
      DISABLED "$<BOOL:${GSL_SKIP_RNG_TEST}>")
  endif()
endfunction()
